/**
 * W-AI-R3（B0 立度量）：困难档决策 + 提示延迟测量与回归门（tsx 运行）。
 *   npm --prefix test exec tsx -- test/bench/ai-latency.ts [--low-end-factor=4]
 *
 * 冻结 v1.9 叫牌/出牌生成分层语料，计时候选 hardStrategy 与 adviseBestMove。
 * decision/hint 是首次遍历（含首次调用）；warm 是相同语料重复三遍，不混合统计。
 * 门＝本机(M4 基线)既有天花板回归：决策 p95<60ms·max<250ms、提示 max<400ms（KNOWN_LIMITATIONS）。
 * 低端机：额外报「实测 × low-end-factor」为代理估计——**需真机校准，不作硬门**（三方审查 #21）。
 */
import os from 'node:os';
import { performance } from 'node:perf_hooks';
import { AiLevel, strategyFor } from '/tmp/ddz-plus5-global54-solver16-tactics-memo2/assets/scripts/ai';
import { adviseBestMove } from '/tmp/ddz-plus5-global54-solver16-tactics-memo2/assets/scripts/ai/hint';
import { buildLatencyCorpus } from '/workspace/danji_doudizhu/test/bench/latency-corpus';

const factorArg = process.argv.find((a) => a.startsWith('--low-end-factor='));
const LOW_END_FACTOR = factorArg ? Number(factorArg.split('=')[1]) : 4;
const WARM_PASSES = 3;

function pct(sorted: number[], q: number): number {
  if (sorted.length === 0) return 0;
  return sorted[Math.min(sorted.length - 1, Math.floor(q * sorted.length))]!;
}
function stats(xs: number[]) {
  const s = xs.slice().sort((a, b) => a - b);
  const round = (v: number) => Math.round(v * 100) / 100;
  return { p50: round(pct(s, 0.5)), p95: round(pct(s, 0.95)), max: round(s[s.length - 1] ?? 0) };
}

async function main(): Promise<void> {
  if (!Number.isFinite(LOW_END_FACTOR) || LOW_END_FACTOR <= 0) throw new Error('low-end-factor must be positive and finite');
  const { frames, hintFrames, metadata } = await buildLatencyCorpus();

  const hard = strategyFor(AiLevel.Hard);
  const decisionMs: number[] = [];
  for (const fx of frames) {
    const t = performance.now();
    hard(fx.snapshot, { next: () => 0 }, fx.history);
    decisionMs.push(performance.now() - t);
  }
  const hintMs: number[] = [];
  for (const fx of hintFrames) {
    const t = performance.now();
    adviseBestMove({ snapshot: fx.snapshot, history: fx.history, viewerSeat: fx.snapshot.turn });
    hintMs.push(performance.now() - t);
  }

  const warmDecisionMs: number[] = [];
  const warmHintMs: number[] = [];
  for (let pass = 0; pass < WARM_PASSES; pass += 1) {
    for (const fx of frames) {
      const t = performance.now();
      hard(fx.snapshot, { next: () => 0 }, fx.history);
      warmDecisionMs.push(performance.now() - t);
    }
    for (const fx of hintFrames) {
      const t = performance.now();
      adviseBestMove({ snapshot: fx.snapshot, history: fx.history, viewerSeat: fx.snapshot.turn });
      warmHintMs.push(performance.now() - t);
    }
  }

  const decision = stats(decisionMs);
  const hint = stats(hintMs);
  const lowEnd = {
    decisionP95: Math.round(decision.p95 * LOW_END_FACTOR * 100) / 100,
    hintMax: Math.round(hint.max * LOW_END_FACTOR * 100) / 100,
  };
  console.log(JSON.stringify({
    frames: frames.length,
    hintFrames: hintFrames.length,
    corpus: metadata,
    environment: {
      node: process.version,
      v8: process.versions.v8,
      platform: process.platform,
      release: os.release(),
      arch: process.arch,
      cpuModel: os.cpus()[0]?.model ?? 'unknown',
      cpuCount: os.cpus().length,
      benchmarkConcurrency: 1,
    },
    measurement: {
      decisionAndHint: 'first traversal; no candidate warm-up; includes first call',
      firstCall: 'first candidate decision after frozen corpus generation; shared rules already exercised',
      firstHint: 'first hint after the decision traversal; shared candidate code already exercised',
      warm: `${WARM_PASSES} repeated traversals of the identical corpus, after both first traversals`,
      gate: 'first-traversal decision/hint only; historical M4 ceilings, host results are not mobile measurements',
      percentile: 'sorted[min(n-1, floor(q*n))], retained for historical compatibility',
    },
    firstCallMs: { decision: Number(decisionMs[0]!.toFixed(2)), hint: Number(hintMs[0]!.toFixed(2)) },
    decision,
    hint,
    warm: {
      passes: WARM_PASSES,
      decisionCalls: warmDecisionMs.length,
      hintCalls: warmHintMs.length,
      decision: stats(warmDecisionMs),
      hint: stats(warmHintMs),
    },
    lowEndFactor: LOW_END_FACTOR,
    lowEnd,
    lowEndIsEstimate: true,
  }, null, 1));

  const reasons: string[] = [];
  if (decision.p95 >= 60) reasons.push(`决策 p95 回归: ${decision.p95}ms ≥ 60ms`);
  if (decision.max >= 250) reasons.push(`决策 max 回归: ${decision.max}ms ≥ 250ms`);
  if (hint.max >= 400) reasons.push(`提示 max 回归: ${hint.max}ms ≥ 400ms`);
  if (reasons.length > 0) {
    console.error(`[latency] GATE FAIL (M4 基线天花板): ${reasons.join(' ; ')}`);
    process.exitCode = 1;
  } else {
    console.log('[latency] PASS (M4 基线天花板)；低端代理见 lowEnd，需真机校准');
  }
}

main().catch((error: unknown) => {
  console.error(error);
  process.exitCode = 1;
});
